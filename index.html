function filling(Z){
let left=Z, arr=[];
for(const [orb,cap,shell] of order){
if(left<=0)break;
const n=Math.min(cap,left); arr.push({orb,n,cap,shell}); left-=n;
}
return arr;
}
function activeDiag(orb){
return diagonals.findIndex(g=>g.includes(orb));
}
function render(){

const [sym,name]=elements[Z-1];

document.querySelectorAll(".occ").forEach(s=>s.textContent="");
f.forEach(x=>{
const n=document.getElementById("n-"+x.orb);
const occ=document.getElementById("occ-"+x.orb);
if(occ) occ.textContent=x.n;
if(n)n.classList.add(x.orb===current.orb?"active":"done");
});
const ad=activeDiag(current.orb);
document.querySelectorAll(".arrow,.arrowhead").forEach(a=>{
a.classList.remove("done","active");
const d=+a.dataset.diag;
if(d<ad)a.classList.add("done");
if(d===ad)a.classList.add("active");
});
configuration.innerHTML=f.map(x=>`<span class="term ${x.orb===current.orb?'active':''}">${x.orb}<sup>${x.n}</sup></span>`).join(" ");
drawAtom(f,Z);
drawOrbitalBoxes(f);
}

function drawAtom(f,Z){
const atom=document.getElementById("atom"); atom.innerHTML="";
const cx=250,cy=250;
const radii=[70,96,123,150,177,204,232];
// Electrons grouped by orbital type on a section of each shell:
// s together, then p, d and f, each with its own colour.
const typeClass={s:"s",p:"p",d:"d",f:"f"};
const typeLabel={s:"s",p:"p",d:"d",f:"f"};
const shellData={};
f.forEach(x=>{
const type=x.orb.slice(-1);
if(!shellData[x.shell]) shellData[x.shell]={s:0,p:0,d:0,f:0};
shellData[x.shell][type]+=x.n;
});
const nuc=document.createElementNS(NS,"circle");
nuc.setAttribute("cx",cx);nuc.setAttribute("cy",cy);nuc.setAttribute("r",51);
nuc.setAttribute("class","nucleus");atom.appendChild(nuc);
const t=document.createElementNS(NS,"text");
t.setAttribute("x",cx);t.setAttribute("y",cy-7);t.setAttribute("text-anchor","middle");
t.setAttribute("fill","white");t.setAttribute("font-size","27");t.setAttribute("font-weight","800");
t.textContent=elements[Z-1][0];
atom.appendChild(t);
Object.keys(shellData).forEach((s,idx)=>{
const r=radii[idx];
const c=document.createElementNS(NS,"circle");
c.setAttribute("cx",cx);c.setAttribute("cy",cy);c.setAttribute("r",r);
c.setAttribute("class","shell");atom.appendChild(c);
// Occupied orbital groups are placed in compact, contiguous arcs.
// This makes all s electrons sit together, all p together, etc.
const groups=["s","p","d","f"].filter(type=>shellData[s][type]>0);
const arcStart=-Math.PI/2;
const groupGap=0.10;
const electronGap=Math.max(0.05, Math.min(0.21, 12/r));
let angle=arcStart;
groups.forEach((type,gi)=>{
const count=shellData[s][type];
for(let j=0;j<count;j++){
const a=angle+j*electronGap;
const e=document.createElementNS(NS,"circle");
e.setAttribute("cx",cx+r*Math.cos(a));
e.setAttribute("cy",cy+r*Math.sin(a));
e.setAttribute("r",idx<3?4.8:4);
e.setAttribute("class","electron "+typeClass[type]);
atom.appendChild(e);
}
angle += Math.max(count,1)*electronGap + groupGap;
});
});
const totalByShell={};
Object.keys(shellData).forEach(s=>{
totalByShell[s]=Object.values(shellData[s]).reduce((a,b)=>a+b,0);
});
// Z dentro del núcleo, bajo el símbolo: nunca tapa electrones.
const zText=document.createElementNS(NS,"text");
zText.setAttribute("x",cx); zText.setAttribute("y",cy+24);
zText.setAttribute("text-anchor","middle");
zText.setAttribute("font-size",19); zText.setAttribute("font-weight",800);
zText.setAttribute("fill","#bfdbfe"); zText.textContent="Z = "+Z;
atom.appendChild(zText);
shellText.innerHTML="Capas: "+Object.keys(totalByShell).map(s=>`${s}ª: ${totalByShell[s]} e⁻`).join(" · ")
+ `<br><span style="display:inline-flex;gap:13px;flex-wrap:wrap;justify-content:center;margin-top:8px">
<span><i class="dot" style="background:#2563eb"></i>s</span>
<span><i class="dot" style="background:#ef4444"></i>p</span>
<span><i class="dot" style="background:#10b981"></i>d</span>
<span><i class="dot" style="background:#a855f7"></i>f</span>
</span>`;
}
